import sys

import os

import re

import json

import requests

from bs4 import BeautifulSoup

import html2text

def clean_filename(title):

    """清理文件名中的非法字符"""

    return re.sub(r'[\\/:*?"&lt;&gt;|]', '_', title).strip()

def clean_html_content(content_soup):

    """

    清理正文中的广告、公众号卡片、赞赏区，并根据关键字切断文末推送内容

    """

    # 1. 移除微信/知乎常见的标准广告与名片组件

    unwanted_selectors = [

        '.js_ad_area', '#js_ad_area',  # 微信底部广告

        '#js_profile_qrcode', '.qr_code_pc',  # 微信二维码

        'mp-common-profile', '.mp-common-profile',  # 微信公众号名片卡片

        '.rich_media_tool', '#js_to_user_profile',  # 关注/阅读原文工具栏

        '.reward_area', '#js_reward_area',  # 赞赏区域

        '.rich_media_area_extra',  # 微信底部推荐区

        '[class*="Card"]', '[class*="McnLink"]',  # 知乎/微信营销卡片

        '[class*="LinkCard"]'

    ]

    for selector in unwanted_selectors:

        for tag in content_soup.select(selector):

            tag.decompose()

    # 2. 尾部截断机制：遇到以下关键字时，直接斩断该节点及其后面的所有 DOM 节点

    cutoff_keywords = [

        '搜索公众号添加',

        '扫码关注',

        '往期推荐',

        '推荐阅读',

        '往期精彩',

        '长按识别二维码',

        '送礼物',

        '发布于'

    ]

    for tag in content_soup.find_all(['p', 'section', 'div', 'h2', 'h3', 'span', 'strong']):

        text = tag.get_text(strip=True)

        # 判断节点文本较短（避免误删大容器）且包含截断关键字

        if len(text) &lt; 100 and any(kw in text for kw in cutoff_keywords):

            print(f"检测到尾部推送标记 [{text[:15]}...]，正在切断后续所有内容...")

            # 获取该节点之后的所有 HTML 元素并彻底删除

            next_elements = list(tag.find_all_next())

            for elem in next_elements:

                elem.decompose()

            # 删除当前节点本身

            tag.decompose()

            break

    # 3. 清理残余的二维码图片（通过 URL 特征过滤）

    for img in content_soup.find_all('img'):

        src = (img.get('data-src') or img.get('src') or '').lower()

        if any(k in src for k in ['qrcode', 'qr_code']):

            img.decompose()

    return content_soup

def download_and_process_images(content_soup, safe_title, base_dir, headers):

    """

    下载文章中的所有图片，保存到 notes/文件夹A/文件夹A 子文件夹中，

    并修改 HTML 中的引用路径为相对路径

    """

    img_dir = os.path.join(base_dir, safe_title)

    os.makedirs(img_dir, exist_ok=True)

    img_tags = content_soup.find_all('img')

    img_count = 0

    for img in img_tags:

        img_url = img.get('data-src') or img.get('src') or img.get('data-original')

        if not img_url or not img_url.startswith('http'):

            continue

        img_count += 1

        img_filename = f"img_{img_count:02d}.png"

        img_save_path = os.path.join(img_dir, img_filename)

        try:

            print(f"正在下载图片 [{img_count}]: {img_url}")

            img_resp = requests.get(img_url, headers=headers, timeout=15)

            if img_resp.status_code == 200:

                with open(img_save_path, 'wb') as f:

                    f.write(img_resp.content)

                # 更新 Markdown 内引用的相对路径

                relative_path = f"{safe_title}/{img_filename}"

                img['src'] = relative_path

                # 清理多余属性

                for attr in ['data-src', 'data-original', 'srcset', 'style']:

                    if attr in img.attrs:

                        del img.attrs[attr]

            else:

                print(f"图片下载失败 (HTTP {img_resp.status_code}): {img_url}")

        except Exception as e:

            print(f"下载图片发生异常: {e}")

    return content_soup

def parse_zhihu(soup, url):

    """专门针对知乎专栏/回答的解析器"""

    author = "Unknown"

    content_html = ""

    script_tag = soup.find('script', id='js-initialData')

    if script_tag and script_tag.string:

        try:

            data = json.loads(script_tag.string)

            article_id = url.split('/')[-1].split('?')[0]

            articles = data.get('initialState', {}).get('entities', {}).get('articles', {})

            if article_id in articles:

                art = articles[article_id]

                author = art.get('author', {}).get('name', author)

                content_html = art.get('content', '')

                print("Successfully parsed Zhihu article via initialData JSON!")

        except Exception as e:

            print(f"JSON parse fallback failed: {e}")

    if not content_html:

        author_node = soup.find('meta', {'itemprop': 'name'}) or soup.find('div', class_='AuthorInfo-name')

        if author_node:

            author = author_node.get('content') or author_node.get_text(strip=True)

        content_node = soup.find('div', class_='Post-RichTextContainer') or \

                       soup.find('div', class_='RichText') or \

                       soup.find('article')

        if content_node:

            content_html = str(content_node)

    return author, content_html

def parse_article(url, custom_title):

    headers = {

        'User-Agent': 'Mozilla/5.0 (iPhone; CPU iPhone OS 16_6 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/16.6 Mobile/15E148 Safari/604.1',

        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',

        'Accept-Language': 'zh-CN,zh-Hans;q=0.9',

    }

    print(f"Requesting: {url}")

    response = requests.get(url, headers=headers, timeout=15)

    print(f"Response Status Code: {response.status_code}")

    if response.status_code != 200:

        print(f"Error: HTTP {response.status_code} returned by target site.")

        sys.exit(1)

    response.encoding = 'utf-8'

    soup = BeautifulSoup(response.text, 'html.parser')

    author = "Unknown"

    content_html = ""

    if "zhihu.com" in url:

        author, content_html = parse_zhihu(soup, url)

    elif "mp.weixin.qq.com" in url:

        author_node = soup.find('a', id='js_name') or soup.find('span', class_='profile_meta_value')

        if author_node:

            author = author_node.get_text(strip=True)

        content_node = soup.find('div', class_='rich_media_content') or soup.find('div', id='js_content')

        if content_node:

            content_html = str(content_node)

    else:

        content_node = soup.find('article') or soup.find('main')

        content_html = str(content_node) if content_node else ""

    if not content_html:

        print(f"Error: Could not extract content HTML from {url}")

        sys.exit(1)

    safe_title = clean_filename(custom_title)

    base_dir = os.path.join("notes", safe_title)

    os.makedirs(base_dir, exist_ok=True)

    # 1. 解析 DOM

    content_soup = BeautifulSoup(content_html, 'html.parser')

    # 2. 执行【尾部截断与净化】（在此步就会彻底砍掉“搜索添加公众号”及之后的所有内容）

    content_soup = clean_html_content(content_soup)

    # 3. 下载保留下来的正文图片（广告与引流图因已被删除，不会被下载）

    content_soup = download_and_process_images(content_soup, safe_title, base_dir, headers)

    # 4. HTML 转 Markdown

    h = html2text.HTML2Text()

    h.ignore_links = False

    h.ignore_images = False

    h.body_width = 0

    markdown_body = h.handle(str(content_soup))

    markdown_content = f"""---

title: "{custom_title}"

author: "{author}"

source_url: "{url}"

---

# {custom_title}

&gt; **作者:** {author}  

&gt; **原文链接:** [{url}]({url})

---

{markdown_body}

"""

    return safe_title, base_dir, markdown_content

def main():

    if len(sys.argv) &lt; 3:

        print("Usage: python fetch_wechat.py &lt;url&gt; &lt;custom_title&gt;")

        sys.exit(1)

    url = sys.argv[1]

    custom_title = sys.argv[2]

    safe_title, base_dir, md_content = parse_article(url, custom_title)

    filepath = os.path.join(base_dir, f"{safe_title}.md")

    with open(filepath, 'w', encoding='utf-8') as f:

        f.write(md_content)

    print(f"\n成功完成！Markdown 文件已保存至: {filepath}")

if __name__ == "__main__":

    main()
