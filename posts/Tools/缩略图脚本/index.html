缩略图脚本

发表于 2026-09-20 分类于 Tools

对大尺寸图像，批量转化为缩略图的 python 脚本


# -*- coding: utf-8 -*-<br>"""<br>将远程 BMP 图像批量生成 PNG 缩略图并导出到本地。<br>远程目录只读，本脚本只读取远程、写入本地输出目录。<br>支持从 config.txt 中按奇偶行读取多组 (REMOTE_DIR, DATE_NAME) 批量处理。<br>"""<br>import os<br>import sys<br>from html import escape<br>from concurrent.futures import ThreadPoolExecutor, as_completed<br>from PIL import Image<br># ---------- 配置 ----------<br>CONFIG_FILE = "config.txt"      # 配置文件路径（奇数行 REMOTE_DIR，偶数行 DATE_NAME）<br>THUMB_WIDTH = 400              # 缩略图宽度（px），高度等比<br>MAX_WORKERS = 8                # 多线程线程数（加速网络传输读取）<br>OUTPUT_ROOT = os.path.dirname(os.path.abspath(__file__))<br># ---------------------------<br>def load_config(config_path):<br>    """读取 txt 配置文件，奇数行为 REMOTE_DIR，偶数行为 DATE_NAME"""<br>    if not os.path.exists(config_path):<br>        print(f"[错误] 找不到配置文件: {config_path}")<br>        print("请在脚本同目录下创建 config.txt，格式如下：")<br>        print("第一行 (REMOTE_DIR): \\\\server\\share\\path1")<br>        print("第二行 (DATE_NAME) : 20250915")<br>        sys.exit(1)<br>    with open(config_path, "r", encoding="utf-8") as f:<br>        lines = [line.strip() for line in f if line.strip() and not line.strip().startswith("#")]<br>    if len(lines) % 2 != 0:<br>        print("[警告] 配置文件有效行数为奇数，最后一行缺失配对的 DATE_NAME，将被忽略！")<br>        lines = lines[:-1]<br>    tasks_config = []<br>    for i in range(0, len(lines), 2):<br>        remote_dir = lines[i]<br>        date_name = lines[i + 1]<br>        tasks_config.append((remote_dir, date_name))<br>    return tasks_config<br>def process_single_image(args):<br>    """单张图片处理逻辑，供线程池并发调用"""<br>    src, out_path, thumb_rel_path, thumb_width = args<br>    try:<br>        with open(src, "rb") as fh:<br>            img = Image.open(fh)<br>            img.load()<br>        w, h = img.size<br>        if w &gt; thumb_width:<br>            new_w = thumb_width<br>            new_h = round(h * (thumb_width / w))<br>        else:<br>            new_w, new_h = w, h<br>        img = img.resize((new_w, new_h), Image.LANCZOS)<br>        if img.mode != "RGB":<br>            img = img.convert("RGB")<br>        img.save(out_path, "PNG")<br>        return True, src, thumb_rel_path, None<br>    except Exception as e:<br>        return False, src, None, str(e)<br>def process_batch(remote_dir, date_name):<br>    """处理一组 (REMOTE_DIR, DATE_NAME) 任务"""<br>    print(f"\n==================================================")<br>    print(f"开始处理任务: [{date_name}]")<br>    print(f"远程路径: {remote_dir}")

print(f"==================================================")<br>    out_dir   = os.path.join(OUTPUT_ROOT, date_name)<br>    thumb_rel = "thumbs"<br>    thumb_dir = os.path.join(out_dir, thumb_rel)<br>    os.makedirs(thumb_dir, exist_ok=True)<br>    try:<br>        files = sorted(f for f in os.listdir(remote_dir) if f.lower().endswith(".bmp"))<br>    except FileNotFoundError as e:<br>        print(f"[错误] 无法访问远程目录: {e}")<br>        return

except PermissionError as e:<br>        print(f"[错误] 拒绝访问，请检查权限或网络登录凭据: {e}")<br>        return

total = len(files)<br>    if total == 0:<br>        print("[提示] 该远程目录下未找到 BMP 文件。")<br>        return

print(f"共发现 {total} 个 BMP 文件，启动 {MAX_WORKERS} 线程处理中...")<br>    tasks = []<br>    for name in files:<br>        src      = os.path.join(remote_dir, name)<br>        out_name = os.path.splitext(name)[0] + ".png"<br>        out_path = os.path.join(thumb_dir, out_name)<br>        rel_path = os.path.join(thumb_rel, out_name)<br>        tasks.append((src, out_path, rel_path, THUMB_WIDTH))<br>    ok, fail = 0, 0<br>    records = []

# 使用线程池加速远程文件 IO 读取<br>    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:<br>        futures = [executor.submit(process_single_image, task) for task in tasks]<br>        for i, future in enumerate(as_completed(futures), 1):<br>            success, src, rel_path, err_msg = future.result()<br>            if success:<br>                ok += 1<br>                records.append((src, rel_path))<br>            else:<br>                fail += 1<br>                print(f"  [失败] {os.path.basename(src)}: {err_msg}")<br>            if i % 20 == 0 or i == total:<br>                print(f"  进度 {i}/{total}")<br>    # 按文件名重新排序记录<br>    records.sort(key=lambda x: x[1])<br>    print(f"任务 [{date_name}] 完成：成功 {ok}，失败 {fail}。缩略图目录: {thumb_dir}")<br>    # 生成预览 HTML<br>    html_path = os.path.join(out_dir, "index.html")<br>    make_html(html_path, date_name, records)<br>    print(f"预览页: {html_path}")<br>def make_html(path, date_name, records):<br>    cards = []<br>    for _src, tp in records:<br>        base = os.path.basename(tp)<br>        # 替换斜杠确保网页路径格式正确<br>        web_tp = tp.replace("\\", "/")<br>        cards.append(<br>            f'&lt;a class="card" href="{escape(web_tp)}" target="_blank"&gt;'<br>            f'&lt;img src="{escape(web_tp)}" loading="lazy" alt="{escape(base)}"/&gt;'<br>            f'&lt;div class="cap"&gt;{escape(base)}&lt;/div&gt;&lt;/a&gt;'<br>        )<br>    cards_html = "\n".join(cards)<br>    html = f"""&lt;!DOCTYPE html&gt;<br>&lt;html lang="zh"&gt;<br>&lt;head&gt;<br>&lt;meta charset="utf-8"/&gt;<br>&lt;meta name="viewport" content="width=device-width, initial-scale=1"/&gt;<br>&lt;title&gt;{escape(date_name)} 缩略图预览&lt;/title&gt;<br>&lt;style&gt;<br>body{{font-family:sans-serif;margin:20px;background:#1e1e1e;color:#ddd}}<br>h1{{font-size:18px;margin-bottom:14px}}<br>.grid{{display:grid;grid-template-columns:repeat(auto-fill,minmax(220px,1fr));gap:14px}}<br>.card{{background:#2c2c2c;border:1px solid #444;border-radius:6px;overflow:hidden;text-decoration:none;color:#ddd}}<br>.card img{{width:100%;height:180px;object-fit:contain;background:#000;display:block}}<br>.cap{{font-size:11px;padding:6px 8px;word-break:break-all}}<br>&lt;/style&gt;<br>&lt;/head&gt;<br>&lt;body&gt;<br>&lt;h1&gt;{escape(date_name)} 共 {len(records)} 张&lt;/h1&gt;<br>&lt;div class="grid"&gt;<br>{cards_html}<br>&lt;/div&gt;<br>&lt;/body&gt;<br>&lt;/html&gt;"""<br>    with open(path, "w", encoding="utf-8") as f:<br>        f.write(html)

def main():<br>    config_path = os.path.join(OUTPUT_ROOT, CONFIG_FILE)<br>    tasks_config = load_config(config_path)<br>    print(f"成功载入 {len(tasks_config)} 组处理任务。")<br>    for remote_dir, date_name in tasks_config:<br>        process_batch(remote_dir, date_name)<br>    print("\n[全部完成] 所有配置任务均已处理完毕！")

if __name__ == "__main__":<br>    main()<br>```
